/*
 * DRAFT display nodes for dm1q.dts (NOT merged into dts/dm1q/dm1q.dts -- no panel driver exists yet, docs/09).
 * Append to a COPY of dm1q.dts to compile-check:
 *   cat dts/dm1q/dm1q.dts notes/2026-10-13-display-bringup/dm1q-display-nodes.dts.txt > /tmp/dm1q-disp.dts
 *   cd dts/dm1q && cpp -nostdinc -I ../../../_scratch/linux-mainline/arch/arm64/boot/dts/qcom \
 *       -I ../../../_scratch/linux-mainline/include -undef -x assembler-with-cpp /tmp/dm1q-disp.dts \
 *     | dtc -@ -I dts -O dtb -o /tmp/dm1q-disp.dtb -
 * (the file needs to live next to sm8550.dtsi's include path only through -I, so /tmp is fine)
 *
 * Provenance per value (docs/09 sections 5-6):
 *   GPIOs 125 / 86 / 187 / 20 / 21, supply voltages     downstream DT, quoted in
 *                                                       notes/2026-10-13-display-bringup/dt-panel-*.txt
 *   vdda (L3E) / vdds (L1E)                             downstream fixup tables + same as gts9uwifi and sm8550-qrd
 *   AVDD gate = PM8550 gpio 11                          INFERENCE (pinctrl node name display_panel_avdd_default,
 *                                                       the regulator node itself carries no gpio property)
 *   compatible "samsung,s6e3fac-amb606aw01"             OUR OWN name, no binding exists upstream
 */

		compatible = "regulator-fixed";
		regulator-name = "panel_vdd3";

		regulator-boot-on;

		pinctrl-names = "default";
	};

	panel_vci: regulator-panel-vci {
		compatible = "regulator-fixed";
		regulator-name = "panel_vci";

		compatible = "regulator-fixed";
		regulator-name = "display_panel_avdd";
		regulator-min-microvolt = <5500000>;
		regulator-max-microvolt = <5500000>;
		gpio = <&pm8550_gpios 11 GPIO_ACTIVE_HIGH>;
		enable-active-high;
		regulator-boot-on;
		startup-delay-us = <233>;
	};
};
